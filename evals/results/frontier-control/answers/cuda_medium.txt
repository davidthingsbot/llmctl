__global__ void block_sum(const float* in, float* out, int n)
{
    extern __shared__ float sdata[];

    unsigned int tid = threadIdx.x;
    unsigned int i   = blockIdx.x * blockDim.x + threadIdx.x;

    // Every thread writes a slot: out-of-range threads contribute zero.
    sdata[tid] = (i < (unsigned int)n) ? in[i] : 0.0f;
    __syncthreads();

    // All threads reach every barrier; only the summing half does work.
    for (unsigned int s = blockDim.x >> 1; s > 0; s >>= 1) {
        if (tid < s) {
            sdata[tid] += sdata[tid + s];
        }
        __syncthreads();
    }

    if (tid == 0) {
        out[blockIdx.x] = sdata[0];
    }
}
