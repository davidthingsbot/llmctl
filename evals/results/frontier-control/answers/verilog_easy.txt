module edge_detect(input clk, input rst_n, input sig, output reg pulse);

    reg sig_d;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            sig_d <= 1'b0;
            pulse <= 1'b0;
        end else begin
            sig_d <= sig;
            pulse <= sig & ~sig_d;
        end
    end

endmodule
