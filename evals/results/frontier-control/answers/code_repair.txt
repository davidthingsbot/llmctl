def decode_frame(frame: bytes):
    if len(frame) < 5:
        raise ValueError("frame too short")
    payload_len = int.from_bytes(frame[1:3], "little")
    if len(frame) != 5 + payload_len:
        raise ValueError("bad frame length")
    payload = frame[3:3 + payload_len]
    expected = int.from_bytes(frame[3 + payload_len:5 + payload_len], "little")
    actual = crc16(frame[:3 + payload_len])
    if actual != expected:
        raise ValueError("bad crc")
    return frame[0], payload
