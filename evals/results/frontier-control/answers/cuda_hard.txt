__global__ void matmul(const float* A, const float* B, float* C,
                       int M, int N, int K)
{
    __shared__ float As[16][16];
    __shared__ float Bs[16][16];

    const int tx  = threadIdx.x;
    const int ty  = threadIdx.y;
    const int row = blockIdx.y * 16 + ty;   // row of C / A
    const int col = blockIdx.x * 16 + tx;   // col of C / B

    float acc = 0.0f;
    const int numTiles = (K + 15) / 16;

    for (int t = 0; t < numTiles; ++t) {
        const int aCol = t * 16 + tx;       // column of A being staged
        const int bRow = t * 16 + ty;       // row of B being staged

        // Out-of-range tile elements load as zero; no out-of-bounds reads.
        As[ty][tx] = (row < M && aCol < K) ? A[(size_t)row * K + aCol] : 0.0f;
        Bs[ty][tx] = (bRow < K && col < N) ? B[(size_t)bRow * N + col] : 0.0f;

        __syncthreads();                    // every thread reaches this, unconditionally

#pragma unroll
        for (int k = 0; k < 16; ++k)
            acc += As[ty][k] * Bs[k][tx];

        __syncthreads();                    // protect the tile before the next load
    }

    if (row < M && col < N)
        C[(size_t)row * N + col] = acc;
}
